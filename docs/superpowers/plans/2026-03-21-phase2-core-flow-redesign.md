# Phase 2: Core Flow Redesign — Intent / Synthesizer / Ledger

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Decompose the overloaded Plan container into three focused containers — Intent (scope declaration), Synthesizer (fact assembly), Ledger (truth sealing) — each with independent versioning, then update all consumers (reports, invoices, API, Android UI).

**Architecture:** Intent replaces Plan's proposal/confirmation lifecycle with a simpler Draft → Proposed → Confirmed → Superseded flow. Synthesizer is a stateless service that assembles Summary Artifacts from Intent + Jobs + Time. Ledger is an append-only store that seals artifacts with SHA-256 hashes. Reports and invoices read exclusively from Ledger entries. The existing report rendering pipeline (reportAssembler → reportRenderer → reportOutput) is preserved but re-sourced from Ledger instead of PlanSnapshot.

**Tech Stack:** TypeScript (backend), Kotlin/Jetpack Compose (Android), Supabase (persistence)

**Spec:** `docs/superpowers/specs/2026-03-20-core-flow-redesign-design.md` (Phase 2 section)

**Depends on:** Phase 1 Messaging Unification (complete)

```typescript
import { IntentVersion, IntentAuthorityResult } from './types';

export function validateIntentCreation(
  scopeStatement: string,
  parties: string[]
): IntentAuthorityResult {
  if (!scopeStatement || scopeStatement.trim().length === 0) {
    return { valid: false, message: 'Scope statement is required' };
  }
  if (!parties || parties.length === 0) {
    return { valid: false, message: 'At least one party UUID is required' };
  }
  return { valid: true, message: 'Intent creation validated' };
}

export function validateIntentConfirmation(
  version: IntentVersion,
  confirmerId: string
): IntentAuthorityResult {
  if (version.status !== 'proposed') {
    return { valid: false, message: `Cannot confirm Intent in status: ${version.status}. Must be 'proposed'.` };
  }
  if (!confirmerId) {
    return { valid: false, message: 'Confirmation requires a human actor UUID. AI cannot confirm Intent.' };
  }
  return { valid: true, message: 'Intent confirmation validated', intentVersion: version };
}

export function validateIntentVersion(
  newVersion: Partial<IntentVersion>,
  priorVersion: IntentVersion
): IntentAuthorityResult {
  if (!newVersion.supersedes || newVersion.supersedes !== priorVersion.id) {
    return { valid: false, message: 'New version must reference the prior version it supersedes' };
  }
  if (priorVersion.status === 'superseded') {
    return { valid: false, message: 'Cannot supersede an already-superseded version' };
  }
  return { valid: true, message: 'Intent version chain validated' };
}
```

```typescript
import { v4 as uuidv4 } from 'uuid';
import { Intent, IntentVersion, IntentStatus } from './types';
import { validateIntentCreation, validateIntentConfirmation, validateIntentVersion } from './intentAuthority';
import { supabase } from './supabase';

export async function createIntent(
  scopeStatement: string,
  parties: string[],
  createdBy: string,
  intendedJobIds: string[] = []
): Promise<{ intent: Intent; version: IntentVersion } | { error: string }> {
  const validation = validateIntentCreation(scopeStatement, parties);
  if (!validation.valid) return { error: validation.message };

  const intentId = uuidv4();
  const versionId = uuidv4();
  const now = Date.now();

  const version: IntentVersion = {
    id: versionId,
    intentId,
    versionNumber: 1,
    status: 'draft',
    scopeStatement,
    intendedJobIds,
    parties,
    createdAt: now,
    autoGenerated: false,
  };

  const intent: Intent = {
    id: intentId,
    currentVersionId: versionId,
    createdBy,
    createdAt: now,
    updatedAt: now,
  };

  // Persist
  const { error: intentErr } = await supabase.from('intents').insert({
    id: intent.id,
    created_by: intent.createdBy,
    created_at: new Date(now).toISOString(),
    updated_at: new Date(now).toISOString(),
  });
  if (intentErr) return { error: intentErr.message };

  const { error: versionErr } = await supabase.from('intent_versions').insert({
    id: version.id,
    intent_id: version.intentId,
    version_number: version.versionNumber,
    status: version.status,
    scope_statement: version.scopeStatement,
    intended_job_ids: version.intendedJobIds,
    parties: version.parties,
    created_at: new Date(now).toISOString(),
    auto_generated: version.autoGenerated,
  });
  if (versionErr) return { error: versionErr.message };

  // Set current_version_id
  await supabase.from('intents').update({ current_version_id: versionId }).eq('id', intentId);

  return { intent, version };
}

export async function proposeIntent(versionId: string): Promise<IntentVersion | { error: string }> {
  const { data, error } = await supabase
    .from('intent_versions').select('*').eq('id', versionId).single();
  if (error || !data) return { error: 'Intent version not found' };
  if (data.status !== 'draft') return { error: 'Can only propose a draft Intent' };

  await supabase.from('intent_versions')
    .update({ status: 'proposed' }).eq('id', versionId);

  return { ...mapVersionRow(data), status: 'proposed' };
}

export async function confirmIntent(
  versionId: string,
  confirmerId: string
): Promise<IntentVersion | { error: string }> {
  const { data, error } = await supabase
    .from('intent_versions').select('*').eq('id', versionId).single();
  if (error || !data) return { error: 'Intent version not found' };

  const version = mapVersionRow(data);
  const validation = validateIntentConfirmation(version, confirmerId);
  if (!validation.valid) return { error: validation.message };

  const now = new Date().toISOString();
  await supabase.from('intent_versions')
    .update({ status: 'confirmed', confirmed_at: now, confirmed_by: confirmerId })
    .eq('id', versionId);

  return { ...version, status: 'confirmed', confirmedAt: Date.now(), confirmedBy: confirmerId };
}

export async function createNewVersion(
  intentId: string,
  priorVersionId: string,
  scopeStatement: string,
  parties: string[],
  intendedJobIds: string[],
  createdBy: string
): Promise<IntentVersion | { error: string }> {
  const { data: priorData, error: priorErr } = await supabase
    .from('intent_versions').select('*').eq('id', priorVersionId).single();
  if (priorErr || !priorData) return { error: 'Prior version not found' };

  const priorVersion = mapVersionRow(priorData);
  const newVersionId = uuidv4();

  const partialNew = { supersedes: priorVersionId };
  const validation = validateIntentVersion(partialNew, priorVersion);
  if (!validation.valid) return { error: validation.message };

  const now = Date.now();
  const newVersion: IntentVersion = {
    id: newVersionId,
    intentId,
    versionNumber: priorVersion.versionNumber + 1,
    status: 'draft',
    scopeStatement,
    intendedJobIds,
    parties,
    supersedes: priorVersionId,
    createdAt: now,
    autoGenerated: false,
  };

  // Mark prior as superseded
  await supabase.from('intent_versions')
    .update({ status: 'superseded', superseded_by: newVersionId })
    .eq('id', priorVersionId);

  // Insert new version
  await supabase.from('intent_versions').insert({
    id: newVersion.id,
    intent_id: newVersion.intentId,
    version_number: newVersion.versionNumber,
    status: newVersion.status,
    scope_statement: newVersion.scopeStatement,
    intended_job_ids: newVersion.intendedJobIds,
    parties: newVersion.parties,
    supersedes: newVersion.supersedes,
    created_at: new Date(now).toISOString(),
    auto_generated: false,
  });

  // Update Intent's current version
  await supabase.from('intents')
    .update({ current_version_id: newVersionId, updated_at: new Date(now).toISOString() })
    .eq('id', intentId);

  return newVersion;
}

export async function autoGenerateIntent(
  jobIds: string[],
  timeEntryIds: string[],
  actorUuid: string
): Promise<{ intent: Intent; version: IntentVersion } | { error: string }> {
  const jobCount = jobIds.length;
  const scope = jobCount === 1
    ? `Completed single job (${jobIds[0].substring(0, 8)})`
    : `Completed ${jobCount} jobs`;

  const result = await createIntent(scope, [actorUuid], actorUuid, jobIds);
  if ('error' in result) return result;

  // Mark as auto-generated and auto-propose
  await supabase.from('intent_versions')
    .update({ auto_generated: true, status: 'proposed' })
    .eq('id', result.version.id);

  result.version.autoGenerated = true;
  result.version.status = 'proposed';

  return result;
}

function mapVersionRow(row: any): IntentVersion {
  return {
    id: row.id,
    intentId: row.intent_id,
    versionNumber: row.version_number,
    status: row.status,
    scopeStatement: row.scope_statement,
    intendedJobIds: row.intended_job_ids || [],
    parties: row.parties || [],
    confirmedAt: row.confirmed_at ? new Date(row.confirmed_at).getTime() : undefined,
    confirmedBy: row.confirmed_by,
    supersededBy: row.superseded_by,
    supersedes: row.supersedes,
    createdAt: new Date(row.created_at).getTime(),
    autoGenerated: row.auto_generated,
  };
}
```

```typescript
  artifact: SummaryArtifact
): LedgerAuthorityResult {
  if (!artifact.serial) {
    return { valid: false, message: 'Artifact must have a valid serial number' };
  }
  if (!artifact.id) {
    return { valid: false, message: 'Artifact must have a valid ID' };
  }
  // Verify SHA-256 can be computed
  const hash = computeHash(artifact);
  if (!hash) {
    return { valid: false, message: 'Failed to compute SHA-256 hash for artifact' };
  }
  return { valid: true, message: 'Sealing validated' };
}

export function validateAmendment(
  newArtifact: SummaryArtifact,
  priorEntry: LedgerEntry
): LedgerAuthorityResult {
  if (!priorEntry.id) {
    return { valid: false, message: 'Prior Ledger entry must exist' };
  }
  if (priorEntry.supersededBy) {
    return { valid: false, message: 'Prior entry already superseded — cannot amend twice' };
  }
  return { valid: true, message: 'Amendment validated' };
}

export function computeHash(artifact: SummaryArtifact): string {
  const canonical = JSON.stringify({
    serial: artifact.serial,
    intentVersionId: artifact.intentVersionId,
    scopeStatement: artifact.scopeStatement,
    workPerformed: artifact.workPerformed,
    laborRecorded: artifact.laborRecorded,
    totalHours: artifact.totalHours,
    totalCost: artifact.totalCost,
    jobIds: artifact.jobIds.sort(),
    timeEntryIds: artifact.timeEntryIds.sort(),
  });
  return crypto.createHash('sha256').update(canonical).digest('hex');
}
```

```typescript
export async function amend(
  newArtifact: SummaryArtifact,
  priorEntryId: string,
  actorUuid: string
): Promise<LedgerEntry | { error: string }> {
  // Fetch prior entry
  const { data: priorData, error: fetchErr } = await supabase
    .from('ledger_entries').select('*').eq('id', priorEntryId).single();
  if (fetchErr || !priorData) return { error: 'Prior Ledger entry not found' };

  const priorEntry = mapLedgerRow(priorData);
  const validation = validateAmendment(newArtifact, priorEntry);
  if (!validation.valid) return { error: validation.message };

  const hash = computeHash(newArtifact);

  const newEntry: LedgerEntry = {
    id: uuidv4(),
    artifactSerial: newArtifact.serial,
    artifactId: newArtifact.id,
    sha256Hash: hash,
    actorUuid,
    supersedes: priorEntryId,
    sealedAt: Date.now(),
  };

  // Mark prior as superseded
  await supabase.from('ledger_entries')
    .update({ superseded_by: newEntry.id })
    .eq('id', priorEntryId);

  // Insert new entry
  const { error } = await supabase.from('ledger_entries').insert({
    id: newEntry.id,
    artifact_serial: newEntry.artifactSerial,
    artifact_id: newEntry.artifactId,
    sha256_hash: newEntry.sha256Hash,
    actor_uuid: newEntry.actorUuid,
    supersedes: newEntry.supersedes,
    sealed_at: new Date(newEntry.sealedAt).toISOString(),
  });

  if (error) return { error: error.message };
  return newEntry;
}

export async function getLedgerEntry(id: string): Promise<LedgerEntry | null> {
  const { data, error } = await supabase
    .from('ledger_entries').select('*').eq('id', id).single();
  if (error || !data) return null;
  return mapLedgerRow(data);
```

```kotlin
package com.guildofsmiths.trademesh.data

import android.content.SharedPreferences
import com.guildofsmiths.trademesh.ui.plan.*
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import java.security.MessageDigest
import java.util.UUID

object IntentRepository {
    private val _intents = MutableStateFlow<List<IntentData>>(emptyList())
    val intents: StateFlow<List<IntentData>> = _intents

    private val _versions = MutableStateFlow<List<IntentVersionData>>(emptyList())
    val versions: StateFlow<List<IntentVersionData>> = _versions

    private var serialCounter = 0
    private var prefs: SharedPreferences? = null

    fun init(prefs: SharedPreferences) {
        this.prefs = prefs
        serialCounter = prefs.getInt("intent_serial_counter", 0)
    }

    fun createIntent(
        scopeStatement: String,
        parties: List<String>,
        createdBy: String,
        intendedJobIds: List<String> = emptyList()
    ): Pair<IntentData, IntentVersionData> {
        val now = System.currentTimeMillis()
        val intentId = "intent_${now}_${(0..999).random()}"
        val versionId = UUID.randomUUID().toString()

        val version = IntentVersionData(
            id = versionId,
            intentId = intentId,
            versionNumber = 1,
            status = IntentStatus.DRAFT,
            scopeStatement = scopeStatement,
            intendedJobIds = intendedJobIds,
            parties = parties,
            createdAt = now,
            autoGenerated = false
        )

        val intent = IntentData(
            id = intentId,
            currentVersionId = versionId,
            createdBy = createdBy,
            createdAt = now,
            updatedAt = now
        )

        _intents.value = listOf(intent) + _intents.value
        _versions.value = listOf(version) + _versions.value
        return Pair(intent, version)
    }

    fun proposeVersion(versionId: String): Boolean {
        val updated = _versions.value.map {
            if (it.id == versionId && it.status == IntentStatus.DRAFT) {
                it.copy(status = IntentStatus.PROPOSED)
            } else it
        }
        _versions.value = updated
        return updated.any { it.id == versionId && it.status == IntentStatus.PROPOSED }
    }

    fun confirmVersion(versionId: String, confirmerId: String): Boolean {
        val now = System.currentTimeMillis()
        val updated = _versions.value.map {
            if (it.id == versionId && it.status == IntentStatus.PROPOSED) {
                it.copy(status = IntentStatus.CONFIRMED, confirmedAt = now, confirmedBy = confirmerId)
            } else it
        }
        _versions.value = updated
        return updated.any { it.id == versionId && it.status == IntentStatus.CONFIRMED }
    }

    fun createNewVersion(
        intentId: String,
        priorVersionId: String,
        scopeStatement: String,
        parties: List<String>,
        intendedJobIds: List<String>
    ): IntentVersionData? {
        val prior = _versions.value.find { it.id == priorVersionId } ?: return null
        if (prior.status == IntentStatus.SUPERSEDED) return null

        val now = System.currentTimeMillis()
        val newVersionId = UUID.randomUUID().toString()

        val newVersion = IntentVersionData(
            id = newVersionId,
            intentId = intentId,
            versionNumber = prior.versionNumber + 1,
            status = IntentStatus.DRAFT,
            scopeStatement = scopeStatement,
            intendedJobIds = intendedJobIds,
            parties = parties,
            supersedes = priorVersionId,
            createdAt = now,
            autoGenerated = false
        )

        // Supersede prior
        _versions.value = _versions.value.map {
            if (it.id == priorVersionId) it.copy(status = IntentStatus.SUPERSEDED, supersededBy = newVersionId)
            else it
        } + newVersion

        // Update intent's current version
        _intents.value = _intents.value.map {
            if (it.id == intentId) it.copy(currentVersionId = newVersionId, updatedAt = now)
            else it
        }

        return newVersion
    }

    fun getVersionsForIntent(intentId: String): List<IntentVersionData> {
        return _versions.value.filter { it.intentId == intentId }.sortedByDescending { it.versionNumber }
    }

    fun getCurrentVersion(intentId: String): IntentVersionData? {
        val intent = _intents.value.find { it.id == intentId } ?: return null
        return _versions.value.find { it.id == intent.currentVersionId }
    }

    fun generateSerial(): String {
        serialCounter++
        prefs?.edit()?.putInt("intent_serial_counter", serialCounter)?.apply()
        val year = java.util.Calendar.getInstance().get(java.util.Calendar.YEAR)
        return "INT-$year-${String.format("%03d", serialCounter)}"
    }
}
```
