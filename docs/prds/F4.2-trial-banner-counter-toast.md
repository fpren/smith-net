### 3.1 TrialBanner

```kotlin
// android/.../ui/components/TrialBanner.kt
@Composable
fun TrialBanner(navigateToPricing: () -> Unit) {
    val entitlements by EntitlementsRepository.entitlements.collectAsState()
    val ent = entitlements ?: return
    if (!ent.trialing) return
    val daysLeft = ((ent.trialEndsAt ?: 0L) - System.currentTimeMillis()) / 86_400_000L

    val copy = trialCopyFor(ent.tier, daysLeft.toInt(), ent.founderSeatsRemaining)

    Row(
        Modifier.fillMaxWidth()
            .background(ConsoleTheme.surface)
            .clickable { navigateToPricing() }
            .padding(horizontal = 12.dp, vertical = 6.dp)
            .drawBehind {
                drawLine(
                    color = ConsoleTheme.outline,
                    start = Offset(0f, size.height),
                    end = Offset(size.width, size.height),
                    strokeWidth = 1.dp.toPx()
                )
            },
        verticalAlignment = Alignment.CenterVertically
    ) {
        Text(copy, style = ConsoleTheme.captionBold, color = ConsoleTheme.textPrimary)
    }
}

private fun trialCopyFor(tier: Tier, daysLeft: Int, seats: FounderSeatsCounts): String = when {
    tier == Tier.SOLO && daysLeft in 1..7 -> "SOLO TRIAL · ${daysLeft} DAYS LEFT · TAP TO LOCK FOUNDER PRICING"
    tier == Tier.SOLO && daysLeft in 8..12 -> "SOLO TRIAL · ${daysLeft} DAYS LEFT · ${seats.solo} FOUNDER SPOTS LEFT"
    tier == Tier.SOLO && daysLeft in 13..14 -> "SOLO TRIAL ENDS IN ${daysLeft} DAYS · TAP TO STAY SOLO"
    tier == Tier.ADVANCED && daysLeft <= 14 -> "ADVANCED TRIAL · ${daysLeft} DAYS · SMITHAI IS LEARNING YOUR JOBS"
    tier == Tier.ADVANCED && daysLeft in 15..28 -> "ADVANCED TRIAL · ${daysLeft} DAYS LEFT · ${seats.advanced} LIFETIME SPOTS LEFT"
    tier == Tier.ADVANCED && daysLeft in 29..30 -> "ADVANCED TRIAL ENDS IN ${daysLeft} DAYS · TAP TO KEEP SMITHAI"
    tier == Tier.ENTERPRISE && daysLeft in 1..13 -> "ENTERPRISE TRIAL · ${daysLeft} DAYS · INVITE YOUR CREW"
    daysLeft <= 0 -> "TRIAL ENDED · YOU'RE ON OPEN · TAP TO REACTIVATE"
    else -> "TRIAL · ${daysLeft} DAYS LEFT"
}
```

### 3.2 FounderSeatsCounter

```kotlin
// android/.../ui/components/FounderSeatsCounter.kt
@Composable
fun FounderSeatsCounter(bonusId: FounderBonusId) {
    val state by FounderSeatsRepository.flowFor(bonusId).collectAsState(initial = SeatsState.Loading)

    val (dotColor, textColor, label) = when (state) {
        is SeatsState.Loading -> Triple(ConsoleTheme.statusGrey, ConsoleTheme.textMuted, "· · · LOADING SPOTS · · ·")
        is SeatsState.Available -> {
            val remaining = (state as SeatsState.Available).remaining
            val total = (state as SeatsState.Available).total
            val color = when {
                remaining > 100 -> ConsoleTheme.textPrimary
                remaining > 10 -> ConsoleTheme.textMuted
                else -> ConsoleTheme.primary
            }
            Triple(ConsoleTheme.statusGreen, color, "${remaining} OF ${total} ${labelFor(bonusId)} LEFT")
        }
        is SeatsState.Exhausted -> Triple(
            ConsoleTheme.statusGrey, ConsoleTheme.textMuted,
            "0 OF ${(state as SeatsState.Exhausted).total} SPOTS — STANDARD PRICING NOW"
        )
    }

    Row(verticalAlignment = Alignment.CenterVertically) {
        Box(Modifier.size(8.dp).clip(CircleShape).background(dotColor))
        Spacer(Modifier.width(8.dp))
        Text(label, style = ConsoleTheme.body, color = textColor)
    }
}

private fun labelFor(id: FounderBonusId): String = when (id) {
    FounderBonusId.SOLO_FOUNDER_PRICING_LOCK -> "FOUNDER SPOTS"
    FounderBonusId.ADVANCED_LIFETIME_TEMPLATE_LIBRARY -> "LIFETIME SPOTS"
    FounderBonusId.ENTERPRISE_FOUNDER_ANNUAL_PRICING -> "FOUNDER ANNUAL SPOTS"
}
```

### 3.3 FounderSeatsRepository (subscribes to WS)

```kotlin
object FounderSeatsRepository {
    private val flows = ConcurrentHashMap<FounderBonusId, MutableStateFlow<SeatsState>>()
    init {
        WsClient.on("founder_seats_changed") { payload ->
            val id = payload.getString("bonusId")
            refresh(FounderBonusId.fromKey(id))
        }
    }
    fun flowFor(id: FounderBonusId): StateFlow<SeatsState> {
        return flows.getOrPut(id) { MutableStateFlow(SeatsState.Loading).also { _ -> refresh(id) } }
    }
    private fun refresh(id: FounderBonusId) {
        scope.launch {
            val res = api.getFounderSeats(id.key)
            val state = if (res.remaining > 0) SeatsState.Available(res.remaining, res.total) else SeatsState.Exhausted(res.total)
            flows[id]?.value = state
        }
    }
}
```
