| Field | Value |
|---|---|

## 1. Problem

Every upgrade CTA from any lock overlay routes here. Without it, conversion path dead-ends.

## 2. Goal

Full screen rendering all 4 tiers per `WIREFRAME-SPEC §5` + 12 states from `STATE-SPEC §N7`.

## 3. Design

```kotlin
// android/.../ui/subscription/TierPricingScreen.kt
@Composable
fun TierPricingScreen(
    initialScrollToTier: Tier? = null,
    onBack: () -> Unit,
    onUpgradeImmediate: (Tier, Cadence) -> Unit,
    onStartTrial: (Tier) -> Unit,
) {
    val entitlements by EntitlementsRepository.entitlements.collectAsState()
    val seats by FounderSeatsRepository.allCounts.collectAsState()
    var annualToggle by remember { mutableStateOf(false) }
    val scrollState = rememberScrollState()

    val ent = entitlements ?: return Loading()

    // Auto-scroll to target tier on first composition
    LaunchedEffect(initialScrollToTier) {
        if (initialScrollToTier != null) scrollState.animateScrollTo(scrollPositionFor(initialScrollToTier))
    }

    Column(Modifier.fillMaxSize().background(ConsoleTheme.background)) {
        // Header
        Row(Modifier.fillMaxWidth().background(ConsoleTheme.surface).clickable { onBack() }
            .padding(horizontal = 16.dp, vertical = 14.dp)) {
            Text("←", style = ConsoleTheme.title)
            Spacer(Modifier.width(14.dp))
            Text("UPGRADE", style = ConsoleTheme.title)
        }
        ConsoleSeparator()

        Column(Modifier.weight(1f).verticalScroll(scrollState).padding(16.dp)) {
            TierSection(Tier.OPEN, ent, seats, annualToggle, onUpgradeImmediate, onStartTrial)
            ConsoleSeparator()
            TierSection(Tier.SOLO, ent, seats, annualToggle, onUpgradeImmediate, onStartTrial)
            ConsoleSeparator()
            TierSection(Tier.ADVANCED, ent, seats, annualToggle, onUpgradeImmediate, onStartTrial)
            ConsoleSeparator()
            TierSection(Tier.ENTERPRISE, ent, seats, annualToggle, onUpgradeImmediate, onStartTrial)
            ConsoleSeparator()
            AnnualToggle(annualToggle) { annualToggle = it }
            ConsoleSeparator()
            AnchorTable()
        }
    }
}

@Composable
private fun TierSection(
    tier: Tier,
    ent: Entitlements,
    seats: FounderSeatsCounts,
    annual: Boolean,
    onUpgrade: (Tier, Cadence) -> Unit,
    onStartTrial: (Tier) -> Unit,
) {
    val isCurrent = tier == ent.tier
    val isAbove = tier.rank > ent.tier.rank
    val isBelow = tier.rank < ent.tier.rank
    val tint = if (isCurrent) ConsoleTheme.surfaceVariant else Color.Unspecified

    Column(Modifier.fillMaxWidth().background(tint).padding(vertical = 12.dp)) {
        Text(displayNameFor(tier), style = ConsoleTheme.title)
        Text(priceFor(tier, annual), style = ConsoleTheme.bodyBold)
        Spacer(Modifier.height(6.dp))
        Text(heroFor(tier), style = ConsoleTheme.body)
        Spacer(Modifier.height(14.dp))

        Text("WHAT'S INCLUDED:", style = ConsoleTheme.captionBold)
        Spacer(Modifier.height(8.dp))
        includedItemsFor(tier).forEach { item ->
            Row(verticalAlignment = Alignment.CenterVertically) {
                Box(Modifier.size(8.dp).clip(CircleShape).background(ConsoleTheme.statusGreen))
                Spacer(Modifier.width(8.dp))
                Text(item, style = ConsoleTheme.body)
            }
            Spacer(Modifier.height(4.dp))
        }

        if (tier != Tier.OPEN) {
            Spacer(Modifier.height(14.dp))
            Text("BONUSES:", style = ConsoleTheme.captionBold)
            Spacer(Modifier.height(8.dp))
            bonusesFor(tier).forEach { (text, bonusId) ->
                Row { Text("★ $text", style = ConsoleTheme.body) }
                if (bonusId != null) {
                    Spacer(Modifier.height(4.dp))
                    FounderSeatsCounter(bonusId = bonusId)
                }
                Spacer(Modifier.height(6.dp))
            }
        }

        Spacer(Modifier.height(14.dp))
        when {
            isCurrent && ent.trialing -> Text("[ TRIAL — ${daysLeft(ent.trialEndsAt)} DAYS LEFT ]", color = ConsoleTheme.primary)
            isCurrent -> Text("[ CURRENT TIER ]", color = ConsoleTheme.textMuted)
            isAbove -> Column {
                if (tier != Tier.ENTERPRISE) {
                    PrimaryCta("TRY ${displayNameFor(tier).uppercase()} FREE ${trialDaysFor(tier)} DAYS — NO CC") {
                        onStartTrial(tier)
                    }
                    Spacer(Modifier.height(8.dp))
                    SecondaryCta("Start immediately (${priceFor(tier, annual)})") {
                        onUpgrade(tier, if (annual) Cadence.ANNUAL else Cadence.MONTHLY)
                    }
                } else {
                    PrimaryCta("START 14-DAY ENTERPRISE TRIAL") { onStartTrial(tier) }
                }
            }
            isBelow -> SecondaryCta("DOWNGRADE TO ${displayNameFor(tier).uppercase()}") {
                showDowngradeConfirmDialog(tier) { onUpgrade(tier, Cadence.MONTHLY) }
            }
        }
    }
}

@Composable
private fun AnnualToggle(value: Boolean, onChange: (Boolean) -> Unit) {
    Row(verticalAlignment = Alignment.CenterVertically) {
        Text("ANNUAL?  ", style = ConsoleTheme.captionBold)
        Text(if (!value) "[ ●  Monthly  ○  Annual — save 16.7% ]" else "[ ○  Monthly  ●  Annual — save 16.7% ]",
            style = ConsoleTheme.body,
            modifier = Modifier.clickable { onChange(!value) })
    }
}

@Composable
private fun AnchorTable() {
    Column {
        Text("WHY OUR PRICE LOOKS LIKE A TYPO", style = ConsoleTheme.captionBold)
        Spacer(Modifier.height(10.dp))
        listOf(
            "JobTread        \$199/mo",
            "Houzz Pro        \$85/mo",
            "Knowify          \$78/mo",
            "ServiceTitan    \$398/mo",
            "Smith Net Solo  \$2.99/mo",
        ).forEach { Text(it, style = ConsoleTheme.body) }
        Spacer(Modifier.height(10.dp))
        Text("Same problem. Different math.", style = ConsoleTheme.body)
    }
}
```

## 4. Acceptance criteria

| AC | Description |
|---|---|
| AC-1 | All 4 tier sections render in fixed order: Open → Solo → Advanced → Enterprise |
| AC-2 | Current tier section has `surfaceVariant` background tint |
| AC-3 | Current tier shows `[ CURRENT TIER ]` (or trial badge if trialing) |
| AC-4 | Tier above current shows primary trial CTA + secondary immediate-pay CTA |
| AC-5 | Tier below current shows downgrade CTA + confirmation dialog |
| AC-6 | Founder counters render per tier where bonus_id applicable |
| AC-7 | Annual toggle re-renders prices |
| AC-8 | Anchor table shows competitor prices with monospace alignment |
| AC-9 | initialScrollToTier auto-scrolls to that section on open |
| AC-10 | Loading state shows skeleton dashes max 800ms |
| AC-11 | Network unavailable: cached entitlements + disabled CTAs `[ NO CONNECTION — TRY AGAIN ]` |
| AC-12 | TalkBack flow: title → body → bonuses → CTAs in order per section |

## 5. BDD scenarios

```gherkin
Scenario: Solo current user views screen
  Given a paid Solo user
  When they navigate to TierPricingScreen
  Then the Solo section has surfaceVariant tint and shows "[ CURRENT TIER ]"
  And the Open section shows "[ DOWNGRADE TO OPEN ]"
  And Advanced + Enterprise sections show trial CTAs

Scenario: Annual toggle changes prices
  Given the toggle is off (monthly)
  When the user taps the toggle
  Then Solo section shows $29.90/yr (instead of $2.99/mo)
  And Advanced shows $99.90/yr
  And the toggle indicator shows ●  Annual

Scenario: User upgrades from Open to Solo via trial
  Given an Open user
  When they tap "TRY SOLO FREE 14 DAYS — NO CC"
  Then onStartTrial(SOLO) is called
  And TrialService.startTrial fires (per F7.1)
```
